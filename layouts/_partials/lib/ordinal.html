{{- /* Input: a positive integer. Output: it as an English ordinal ("1st", "12th", "23rd"). */ -}} {{- $n := int . -}} {{- $suffix := "th" -}} {{- if not (in (slice 11 12 13) (mod $n 100)) -}} {{- $suffix = index (dict "1" "st" "2" "nd" "3" "rd") (string (mod $n 10)) | default "th" -}} {{- end -}} {{- return (printf "%d%s" $n $suffix) -}}
