{{- /* Output: a dict keyed by "/" (or normalized text, for goals the catalog doesn't know) -> {id, text, game_type, known, boards, marks}. Catalogued goals come from data/square_stats/, the same numbers their own pages show: `boards` is how many games dealt the goal, `marks` how many ended with it claimed. Those files keep base and DLC apart even where a goal's text or id is shared between them, which a text lookup can't. Transcriptions the catalog doesn't recognise have no stats file, so they are still counted here from the match files -- `marks` is then raw mark events. Call through `partialCached`; it scans every match. */ -}} {{- $stats := dict -}} {{- range $gt, $squares := hugo.Data.square_stats -}} {{- range $squares -}} {{- if .num_games -}} {{- $stats = merge $stats (dict (printf "%s/%s" $gt .id) (dict "id" .id "text" .text "game_type" $gt "known" true "boards" .num_games "marks" (.games_marked | default 0))) -}} {{- end -}} {{- end -}} {{- end -}} {{- range $seasonDir, $matchesInSeason := hugo.Data.matches -}} {{- range $file, $match := $matchesInSeason -}} {{- range $g := ($match.games | default slice) -}} {{- $gt := $g.game_type | default "" -}} {{- range $row := ($g.square_texts | default slice) -}} {{- range $cell := ($row | default slice) -}} {{- $sq := partial "lib/square-lookup.html" $cell -}} {{- if and $sq.text (not $sq.known) -}} {{- $key := lower $sq.text -}} {{- $entry := index $stats $key | default (dict "id" "" "text" $sq.text "game_type" $gt "known" false "boards" 0 "marks" 0) -}} {{- $stats = merge $stats (dict $key (merge $entry (dict "boards" (add $entry.boards 1)))) -}} {{- end -}} {{- end -}} {{- end -}} {{- range $g.events -}} {{- if and (eq .event_type "mark") .color -}} {{- $sq := partial "lib/square-lookup.html" .square_text -}} {{- if and $sq.text (not $sq.known) -}} {{- $key := lower $sq.text -}} {{- $entry := index $stats $key | default (dict "id" "" "text" $sq.text "game_type" $gt "known" false "boards" 0 "marks" 0) -}} {{- $stats = merge $stats (dict $key (merge $entry (dict "marks" (add $entry.marks 1)))) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $stats -}}
