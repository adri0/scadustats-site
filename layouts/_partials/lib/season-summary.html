{{- /* Input: a season id. Output: {matches, games, base_games, dlc_games, coop_games, untranscribed, champion, qualified, phases}, counting every match file of the season (lib/season-matches.html, by each match's `season` field) plus the hand-entered (untranscribed) matches of data/brackets/season-.yaml. base_games/dlc_games/coop_games (Season 5's mixed boards) count transcribed games only, since hand-entered matches carry no game types. `champion` is the Grand Final winner (or, in a bracket with no `final`, the winner of the upper-bracket match with no `next`) as a slot (lib/season-slots.html) or player-lookup dict, "" while there isn't one. `qualified` is one {player, to} per decided bracket match with `qualifies` (e.g. pre-season 6's Upper and Lower Final winners, who qualified for Season 6), in file order. `phases` is one {id ("groups"/"playoffs"/"bracket"), name, matches, games, start, end} per stage, round robin, then the bracket's `playoffs` matches, then the rest of the bracket, and empty when the season has neither data/groups nor data/brackets: match files are split by whether the bracket names them, because their `match_type` isn't reliable. */ -}} {{- $sid := printf "%v" . -}} {{- $seasonMatches := partialCached "lib/season-matches.html" $sid $sid -}} {{- $bracket := partial "lib/season-file.html" (dict "kind" "brackets" "season_id" $sid) -}} {{- $groupsCfg := partial "lib/season-file.html" (dict "kind" "groups" "season_id" $sid) -}} {{- $slots := partial "lib/season-slots.html" $sid -}} {{- $playoffKeys := slice -}} {{- with $bracket -}} {{- range .matches }}{{ with .match }}{{ $playoffKeys = $playoffKeys | append . }}{{ end }}{{ end -}} {{- end -}} {{- /* Non-bracket match files: the round robin when there are groups. */ -}} {{- $rr := dict "matches" 0 "games" 0 "dates" slice -}} {{- $total := dict "matches" 0 "games" 0 -}} {{- $base := 0 }}{{ $dlc := 0 }}{{ $coop := 0 -}} {{- range $key, $raw := $seasonMatches -}} {{- $s := partial "lib/match-summary.html" $raw -}} {{- range $raw.games -}} {{- if eq .game_type "base" }}{{ $base = add $base 1 }}{{ else if eq .game_type "dlc" }}{{ $dlc = add $dlc 1 }}{{ else if eq .game_type "2v2" }}{{ $coop = add $coop 1 }}{{ end -}} {{- end -}} {{- $total = dict "matches" (add $total.matches 1) "games" (add $total.games $s.num_games) -}} {{- if not (in $playoffKeys $key) -}} {{- $rr = dict "matches" (add $rr.matches 1) "games" (add $rr.games $s.num_games) "dates" ($rr.dates | append ($s.match_date | default slice)) -}} {{- end -}} {{- end -}} {{- $phases := slice -}} {{- with $groupsCfg -}} {{- $span := partial "lib/date-span.html" $rr.dates -}} {{- $phases = $phases | append (merge $span (dict "id" "groups" "name" (.title | default "Groups") "matches" $rr.matches "games" $rr.games)) -}} {{- end -}} {{- $champion := "" -}} {{- $qualified := slice -}} {{- $untranscribed := 0 -}} {{- with $bracket -}} {{- $results := partial "lib/bracket-results.html" (dict "season_id" $sid "bracket" .) -}} {{- $hasFinal := gt (len (where .matches "bracket" "final")) 0 -}} {{- $po := dict "playoffs" (dict "matches" 0 "games" 0 "dates" slice) "bracket" (dict "matches" 0 "games" 0 "dates" slice) -}} {{- range $results -}} {{- $phase := cond (eq .entry.bracket "playoffs") "playoffs" "bracket" -}} {{- $games := 0 -}} {{- with .entry.match -}} {{- with index $seasonMatches . }}{{ $games = (partial "lib/match-summary.html" .).num_games }}{{ end -}} {{- else -}} {{- /* Hand-entered: only the scores are known, so count decided games. */ -}} {{- if and (ne .red_score nil) (ne .blue_score nil) -}} {{- $games = add .red_score .blue_score -}} {{- $untranscribed = add $untranscribed 1 -}} {{- $total = dict "matches" (add $total.matches 1) "games" (add $total.games $games) -}} {{- end -}} {{- end -}} {{- if or .path $games -}} {{- $p := index $po $phase -}} {{- $po = merge $po (dict $phase (dict "matches" (add $p.matches 1) "games" (add $p.games $games) "dates" ($p.dates | append (.entry.date | default slice)))) -}} {{- end -}} {{- $decider := cond $hasFinal (eq .entry.bracket "final") (and (eq .entry.bracket "upper") (not .entry.next)) -}} {{- if and $decider (in (slice "red" "blue") .winner) -}} {{- $w := cond (eq .winner "red") .red .blue -}} {{- $champion = index $slots $w.slug | default $w -}} {{- end -}} {{- if and .entry.qualifies (in (slice "red" "blue") .winner) -}} {{- $w := cond (eq .winner "red") .red .blue -}} {{- $qualified = $qualified | append (dict "player" (index $slots $w.slug | default $w) "to" .entry.qualifies) -}} {{- end -}} {{- end -}} {{- range $ph := slice (dict "id" "playoffs" "name" "Play-offs") (dict "id" "bracket" "name" "Double elimination") -}} {{- $p := index $po $ph.id -}} {{- if or $p.matches (eq $ph.id "bracket") -}} {{- $span := partial "lib/date-span.html" $p.dates -}} {{- $phases = $phases | append (merge $span $ph (dict "matches" $p.matches "games" $p.games)) -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "matches" $total.matches "games" $total.games "base_games" $base "dlc_games" $dlc "coop_games" $coop "untranscribed" $untranscribed "champion" $champion "qualified" $qualified "phases" $phases) -}}
