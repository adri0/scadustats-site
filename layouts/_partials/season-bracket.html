{{- /* Input: dict {season_id, bracket} where bracket is a data/brackets/season-.yaml entry. Renders the playoff bracket as an inline SVG: one card per match laid out on a col/row grid, with elbow connectors from each match to its `next`. Players, scores and winners come from lib/bracket-results.html. */ -}} {{- $bracket := .bracket -}} {{- /* Geometry, in SVG user units. A card is a header line plus two player rows. */ -}} {{- $cardW := 200 -}} {{- $headH := 18 -}} {{- $rowH := 26 -}} {{- $rowGap := 4 -}} {{- $cardH := add $headH $rowH $rowGap $rowH -}} {{- $midY := add $headH $rowH (div $rowGap 2) -}} {{- $colGap := 56 -}} {{- $rowUnit := 96 -}} {{- $left := 32 -}} {{- $pad := 8 -}} {{- $cards := dict -}} {{- $maxCol := 0 -}} {{- $maxRow := 0.0 -}} {{- range partial "lib/bracket-results.html" . -}} {{- $e := .entry -}} {{- $x := add $left (mul $e.col (add $cardW $colGap)) -}} {{- $y := add $pad (mul (float $e.row) $rowUnit) -}} {{- $maxCol = cond (gt $e.col $maxCol) $e.col $maxCol -}} {{- $maxRow = cond (gt (float $e.row) $maxRow) (float $e.row) $maxRow -}} {{- $cards = merge $cards (dict (string $e.id) (merge . (dict "m" $e "x" $x "y" $y))) -}} {{- end -}} {{- $width := add $left (mul (add $maxCol 1) $cardW) (mul $maxCol $colGap) -}} {{- $height := add $pad (mul $maxRow $rowUnit) $cardH $pad -}}
