A small round avatar sized to the surrounding text; players without one get their initial in a circle so names stay aligned. Purely decorative next to the name, so it's hidden from screen readers. */ -}} {{- $p := .player -}} {{- $class := cond (.end | default false) "avatar avatar-end" "avatar" -}} {{- with $p.avatar -}} {{- else -}} {{ substr ($p.display_name | default "?") 0 1 | upper }} {{- end -}}
